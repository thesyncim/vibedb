package durable

import "testing"

func diagnosticScalarTouchUnifiedScanAllBytes(value []byte) byte {
	var sink byte
	for _, b := range value {
		sink ^= b
	}
	return sink
}

func TestDiagnosticStabilizedTouchUnifiedScanAllBytes(t *testing.T) {
	lengths := []int{0, 1, 2, 3, 7, 8, 9, 15, 16, 17, 31, 32, 63, 64, 65, 127, 128, 129, 255, 256, 257, 1023, 1024, 1025}
	for _, length := range lengths {
		for offset := 0; offset < 64; offset++ {
			data := make([]byte, offset+length+64)
			for i := range data {
				data[i] = byte((i*131 + offset*17 + length*7) ^ (i >> 3))
			}
			data = data[offset : offset+length]
			want := diagnosticScalarTouchUnifiedScanAllBytes(data)
			if got := touchUnifiedScanAllBytes(data); got != want {
				t.Fatalf("length=%d offset=%d got=%d want=%d", length, offset, got, want)
			}
		}
	}
}
