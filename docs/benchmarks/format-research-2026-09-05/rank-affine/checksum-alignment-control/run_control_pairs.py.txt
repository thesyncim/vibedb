#!/usr/bin/env python3
"""Run six alternating control pairs and preserve every raw output."""
import argparse
import hashlib
import json
import os
from pathlib import Path
import platform
import re
import subprocess
import time


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--binary", required=True, type=Path)
    parser.add_argument("--output", required=True, type=Path)
    parser.add_argument("--runs", type=int, default=6)
    parser.add_argument("--benchtime", default="200ms")
    args = parser.parse_args()
    if not 3 <= args.runs <= 30:
        parser.error("runs must be 3..30")
    binary = args.binary.resolve()
    args.output.mkdir(parents=True, exist_ok=False)
    metadata = {
        "binary": str(binary),
        "binary_sha256": hashlib.sha256(binary.read_bytes()).hexdigest(),
        "runs": args.runs,
        "benchtime": args.benchtime,
        "input_lengths": [249, 4096],
        "gomaxprocs": 1,
        "order": "offset24/offset56 odd pairs; offset56/offset24 even pairs",
        "platform": platform.platform(),
        "started_unix_seconds": time.time(),
    }
    (args.output / "environment.json").write_text(json.dumps(metadata, indent=2) + "\n")
    env = dict(os.environ, GOMAXPROCS="1")
    rows = []
    for pair in range(1, args.runs + 1):
        order = ("offset24", "offset56") if pair % 2 else ("offset56", "offset24")
        for offset in order:
            log = args.output / f"pair-{pair:02d}-{offset}.txt"
            command = [
                str(binary),
                "-test.run=^$",
                f"-test.bench=^BenchmarkXorAlignment$/{offset}/",
                f"-test.benchtime={args.benchtime}",
                "-test.count=1",
                "-test.cpu=1",
                "-test.timeout=8m",
            ]
            started = time.monotonic()
            with log.open("w") as stream:
                result = subprocess.run(command, env=env, stdout=stream, stderr=subprocess.STDOUT)
            output = log.read_text()
            names = {
                re.sub(r"-[0-9]+$", "", line.split()[0])
                for line in output.splitlines()
                if line.startswith("Benchmark") and line.split()
            }
            missing = [
                f"BenchmarkXorAlignment/{offset}/bytes={length}"
                for length in (249, 4096)
                if f"BenchmarkXorAlignment/{offset}/bytes={length}" not in names
            ]
            if missing:
                raise RuntimeError(f"{log.name} omitted expected control cases: {missing}")
            row = {
                "pair": pair,
                "offset": offset,
                "command": command,
                "log": log.name,
                "returncode": result.returncode,
                "elapsed_seconds": time.monotonic() - started,
            }
            rows.append(row)
            (args.output / "runs.json").write_text(json.dumps(rows, indent=2) + "\n")
            if result.returncode:
                raise RuntimeError(f"{log.name} failed with {result.returncode}")
            print(f"completed control pair {pair}/{args.runs} {offset}", flush=True)


if __name__ == "__main__":
    main()
