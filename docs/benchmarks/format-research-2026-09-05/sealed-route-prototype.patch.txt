+// routeEncoding is authenticated as part of the fixed route descriptor.
+// The zero value retains the original per-entry payload byte for byte.
+type routeEncoding uint32
+
+const (
+	routeEncodingEntries routeEncoding = iota
+	routeEncodingRuns
+	sealedRouteMaxRuns = 16
+)
+
+type routeExtentRun struct {
+	first routeEntry
+	count uint32
+}
+
+type routeRunPlan struct {
+	runs                     [sealedRouteMaxRuns]routeExtentRun
+	count, bytes, entryBytes int
+}
+
+// appendBestRoutePayload proves a bounded run representation before allocating
+// its payload. Irregular blocks retain the exact original encoding. This runs
+// only while sealing; active wave encoding and the decoded route cache stay
+// unchanged.
+func appendBestRoutePayload(dst []byte, entries []routeEntry) ([]byte, routeEncoding, error) {
+	var plan routeRunPlan
+	if !plan.build(entries) || plan.bytes >= plan.entryBytes {
+		payload, err := appendRoutePayload(dst, entries)
+		return payload, routeEncodingEntries, err
+	}
+	dst = appendUvarint(dst, entries[0].Term)
+	dst = append(dst, byte(entries[0].Type))
+	dst = appendUvarint(dst, uint64(plan.count))
+	previousExtent := uint64(0)
+	for i := 0; i < plan.count; i++ {
+		run := &plan.runs[i]
+		dst = appendUvarint(dst, uint64(run.count))
+		dst = appendUvarint(dst, run.first.ExtentOffset-previousExtent)
+		dst = appendUvarint(dst, run.first.ExtentBytes)
+		dst = appendUvarint(dst, run.first.ExtentID)
+		dst = append(dst, run.first.BatchID[:]...)
+		dst = appendUvarint(dst, run.first.DataOffset)
+		dst = appendUvarint(dst, run.first.DataBytes)
+		previousExtent = run.first.ExtentOffset
+	}
+	return dst, routeEncodingRuns, nil
+}

+func (p *routeRunPlan) build(entries []routeEntry) bool {
+	if len(entries) < 2 || len(entries) > sealedDefaultBlockEntries {
+		return false
+	}
+	*p = routeRunPlan{}
+	first := entries[0]
+	previousExtent, previousDataEnd := uint64(0), uint64(0)
+	for i := range entries {
+		entry := entries[i]
+		if !validRouteEntry(entry) || entry.Term != first.Term || entry.Type != first.Type || entry.DataBytes == 0 {
+			return false
+		}
+		if i == 0 || entry.ExtentOffset != previousExtent {
+			if p.count == sealedRouteMaxRuns {
+				return false
+			}
+			if i != 0 && (entry.ExtentOffset <= previousExtent || p.runs[p.count-1].first.ExtentBytes > entry.ExtentOffset-previousExtent) {
+				return false
+			}
+			p.runs[p.count] = routeExtentRun{first: entry, count: 1}
+			p.count++
+			p.entryBytes += uvarintBytes(entry.ExtentOffset-previousExtent) + uvarintBytes(entry.ExtentBytes) + uvarintBytes(entry.ExtentID) + len(entry.BatchID)
+			previousExtent = entry.ExtentOffset
+		} else {
+			run := &p.runs[p.count-1]
+			if entry.ExtentBytes != run.first.ExtentBytes || entry.ExtentID != run.first.ExtentID || entry.BatchID != run.first.BatchID || entry.DataBytes != run.first.DataBytes || entry.DataOffset != previousDataEnd {
+				return false
+			}
+			run.count++
+		}
+		// One flags byte and the zero term delta, except for the first term.
+		p.entryBytes += 2 + uvarintBytes(entry.DataOffset) + uvarintBytes(entry.DataBytes)
+		previousDataEnd = entry.DataOffset + entry.DataBytes // validRouteEntry bounded both.
+	}
+	firstTerm, deltaOK := signedDelta(0, first.Term)
+	if !deltaOK {
+		firstTerm = first.Term
+	}
+	p.entryBytes += uvarintBytes(firstTerm) - 1
+	p.bytes = uvarintBytes(first.Term) + 1 + uvarintBytes(uint64(p.count))
+	previousExtent = 0
+	for i := 0; i < p.count; i++ {
+		run := &p.runs[i]
+		p.bytes += uvarintBytes(uint64(run.count)) + uvarintBytes(run.first.ExtentOffset-previousExtent) + uvarintBytes(run.first.ExtentBytes) + uvarintBytes(run.first.ExtentID) + len(run.first.BatchID) + uvarintBytes(run.first.DataOffset) + uvarintBytes(run.first.DataBytes)
+		previousExtent = run.first.ExtentOffset
+	}

+func decodeRoutePayloadEncoding(src []byte, count uint32, dst []routeEntry, encoding routeEncoding) ([]routeEntry, error) {
+	switch encoding {
+	case routeEncodingEntries:
+		return decodeRoutePayload(src, count, dst)
+	case routeEncodingRuns:
+		return decodeRouteRuns(src, count, dst)
+	default:
+		return nil, ErrCorrupt
+	}

+// decodeRouteRuns expands each extent into the existing decoded-cache layout.
+// It validates the entire run before multiplication or writing any entry, so
+// a bad count/stride cannot overflow or manufacture an out-of-extent route.
+func decodeRouteRuns(src []byte, count uint32, dst []routeEntry) ([]routeEntry, error) {
+	if count < 2 || count > sealedDefaultBlockEntries || cap(dst) < int(count) || len(src) > sealedMaxBlockRouteBytes {
+		return nil, ErrCorrupt
+	}
+	cursor := canonicalCursor{data: src}
+	term, err := cursor.uvarint()
+	if err != nil || term == 0 {
+		return nil, ErrCorrupt
+	}
+	kind, err := cursor.byte()
+	if err != nil || kind > 2 {
+		return nil, ErrCorrupt
+	}
+	runs, err := cursor.uvarint()
+	if err != nil || runs == 0 || runs > sealedRouteMaxRuns || runs > uint64(count) {
+		return nil, ErrCorrupt
+	}
+	dst = dst[:count]
+	written := 0
+	previousExtent, previousExtentBytes := uint64(0), uint64(0)
+	for i := uint64(0); i < runs; i++ {
+		entries, err := cursor.uvarint()
+		if err != nil || entries == 0 || entries > uint64(len(dst)-written) {
+			return nil, ErrCorrupt
+		}
+		delta, err := cursor.uvarint()
+		if err != nil || delta > sealedMaxSegmentBytes-previousExtent || i != 0 && delta < previousExtentBytes {
+			return nil, ErrCorrupt
+		}
+		extentOffset := previousExtent + delta
+		extentBytes, err := cursor.uvarint()
+		if err != nil || extentBytes > sealedMaxSegmentBytes-extentOffset {
+			return nil, ErrCorrupt
+		}
+		extentID, err := cursor.uvarint()
+		if err != nil {
+			return nil, ErrCorrupt
+		}
+		batch, err := cursor.take(16)
+		if err != nil {
+			return nil, ErrCorrupt
+		}
+		dataOffset, err := cursor.uvarint()
+		if err != nil || dataOffset > extentBytes {
+			return nil, ErrCorrupt
+		}
+		dataBytes, err := cursor.uvarint()
+		if err != nil || dataBytes == 0 || entries > (extentBytes-dataOffset)/dataBytes {
+			return nil, ErrCorrupt
+		}
+		entry := routeEntry{Term: term, Type: uint16(kind), ExtentOffset: extentOffset, ExtentBytes: extentBytes, ExtentID: extentID, DataOffset: dataOffset, DataBytes: dataBytes}
+		copy(entry.BatchID[:], batch)
+		for j := uint64(0); j < entries; j++ {
+			dst[written] = entry
+			written++
+			entry.DataOffset += dataBytes
+		}
+		previousExtent, previousExtentBytes = extentOffset, extentBytes
+	}
+	if written != len(dst) || !cursor.empty() {
+		return nil, ErrCorrupt
+	}
+	return dst, nil
